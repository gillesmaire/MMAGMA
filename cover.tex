\definecolor{myblue}{RGB}{10,20,60}
\definecolor{mywhite}{RGB}{255,245,200}

    {\coverfont\fontsize{120}{80}\selectfont \@title \par}

{\coverfont\fontsize{28}{34}\selectfont \@author \par}
\vspace{1cm}

{\coverfont\fontsize{18}{22}\selectfont \@date \par}

  \end{center}
  \vspace*{\fill}

  \end{titlepage}
  \nopagecolor
  \color{black}
}
\makeatother
